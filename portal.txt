    PROCEDURE MiniStatement@1000000001(MemberNo@1000000000 : Text[100]) MiniStmt : Text;
    VAR
      minimunCount@1000000002 : Integer;
      amount@1000000003 : Decimal;
    BEGIN
        BEGIN
        MiniStmt :='';
          objMember.RESET;
         objMember.SETRANGE("No."  ,MemberNo);

        IF objMember .FIND('-') THEN BEGIN

           minimunCount:=1;
           //Vendor.CALCFIELDS(Vendor.Balance);
           VendorLedgEntry.SETCURRENTKEY(VendorLedgEntry."Entry No.");
           VendorLedgEntry.ASCENDING(FALSE);
           VendorLedgEntry.SETRANGE(VendorLedgEntry."Customer No.",MemberNo);
           VendorLedgEntry.SETRANGE(VendorLedgEntry.Reversed,FALSE);
          IF VendorLedgEntry.FINDSET THEN BEGIN
              MiniStmt:='';
              REPEAT
                amount:=VendorLedgEntry.Amount;
                IF amount<1 THEN   amount:= amount*-1;
                    MiniStmt :=MiniStmt + FORMAT(VendorLedgEntry."Posting Date") +':::'+ COPYSTR(FORMAT(VendorLedgEntry."Transaction Type"),1,25) +':::' +
                    FORMAT(amount)+'::::';
                    minimunCount:= minimunCount +1;
                    IF minimunCount > 15 THEN BEGIN
                    EXIT(MiniStmt);
                    END
                UNTIL VendorLedgEntry.NEXT =0;
                END;

        END;

        END;
        EXIT(MiniStmt);
    END;

    PROCEDURE FnloanCalc@24(LoanAmount@1000 : Decimal;RepayPeriod@1001 : Integer;LoanCode@1002 : Code[30]) text : Text;
    BEGIN
       {Loansetup.RESET;
             Loansetup.SETRANGE(Code, LoanCode);
             IF Loansetup.FIND('-') THEN BEGIN
              //....................Amortized......................................................//
             IF  Loansetup."Repayment Method"= Loansetup."Repayment Method"::Amortised THEN BEGIN
               Date:=TODAY;
                PRAmount:=LoanAmount;
                text:=text+FORMAT(Date)+'!!'+FORMAT(ROUND( LPrincipal))+'!!'+FORMAT(ROUND( LInterest))+'!!'+FORMAT(ROUND(TotalMRepay))+'!!'+FORMAT(ROUND(LoanAmount)) +'!!'+FORMAT(ROUND(PRAmount))+'??';
              Date:=CALCDATE('+1M', Date);
               REPEAT
                 InterestRate:=Loansetup."Interest rate";
                 TotalMRepay:=ROUND((InterestRate/12/100) / (1 - POWER((1 + (InterestRate/12/100)),- RepayPeriod)) * LoanAmount,0.05,'>');
                  LInterest:=ROUND(LoanAmount / 100 / 12 * InterestRate,0.05,'>');
                  LPrincipal:=TotalMRepay-LInterest;
              PRAmount:=LoanAmount;
                LoanAmount:=LoanAmount-LPrincipal;
             RepayPeriod:= RepayPeriod-1;

              text:=text+FORMAT(Date)+'!!'+FORMAT(ROUND( LPrincipal))+'!!'+FORMAT(ROUND( LInterest))+'!!'+FORMAT(ROUND(TotalMRepay))+'!!'+FORMAT(ROUND(LoanAmount)) +'!!'+FORMAT(ROUND(PRAmount))+'??';
              Date:=CALCDATE('+1M', Date);
              UNTIL RepayPeriod=0;
              END;
              ///.......................end of Amortized..............................................//


              //......................Reducing Balance...............................................//
             IF  Loansetup."Repayment Method"= Loansetup."Repayment Method"::"Reducing Balance" THEN BEGIN
                     Date:=TODAY;
                      PRAmount:=LoanAmount;
                      text:=text+FORMAT(Date)+'!!'+FORMAT(ROUND( LPrincipal))+'!!'+FORMAT(ROUND( LInterest))+'!!'+FORMAT(ROUND(TotalMRepay))+'!!'+FORMAT(ROUND(LoanAmount)) +'!!'+FORMAT(ROUND(PRAmount))+'??';
                    Date:=CALCDATE('+1M', Date);
                     REPEAT
                       InterestRate:=Loansetup."Interest rate";
                       LPrincipal:=ROUND(LoanAmount/RepayPeriod,0.05,'>');
                     LInterest:=ROUND((InterestRate/12/100)*LoanAmount,0.05,'>');
                    TotalMRepay:=LPrincipal+LInterest;
                    PRAmount:=LoanAmount;
                      LoanAmount:=LoanAmount-LPrincipal;
                   RepayPeriod:= RepayPeriod-1;

                    text:=text+FORMAT(Date)+'!!'+FORMAT(ROUND( LPrincipal))+'!!'+FORMAT(ROUND( LInterest))+'!!'+FORMAT(ROUND(TotalMRepay))+'!!'+FORMAT(ROUND(LoanAmount)) +'!!'+FORMAT(ROUND(PRAmount))+'??';
                    Date:=CALCDATE('+1M', Date);
                    UNTIL RepayPeriod=0;
                    END;
                    //....................end of Reducing Balance.......................................////

                    //...........................Constant...............................................////
                    IF  Loansetup."Repayment Method"= Loansetup."Repayment Method"::Constants THEN BEGIN
                    LoansRec.TESTFIELD(LoansRec.Repayment);
                    IF LBalance < LoansRec.Repayment THEN
                    LPrincipal:=LBalance
                    ELSE
                    LPrincipal:=LoansRec.Repayment;
                    LInterest:=LoansRec.Interest;
                    END;

              //.......................Straight Line......................................................///

              IF  Loansetup."Repayment Method"= Loansetup."Repayment Method"::"Straight Line" THEN BEGIN

              {LoansRec.TESTFIELD(LoansRec.Interest);
              LoansRec.TESTFIELD(LoansRec.Installments);
              LPrincipal:=LoanAmount/RepayPeriod;
              LInterest:=(Loansetup."Interest rate"/12/100)*LoanAmount;///RepayPeriod}
                    Date:=TODAY;
                    PRAmount:=LoanAmount;
                    text:=text+FORMAT(Date)+'!!'+FORMAT(ROUND( LPrincipal))+'!!'+FORMAT(ROUND( LInterest))+'!!'+FORMAT(ROUND(TotalMRepay))+'!!'+FORMAT(ROUND(LoanAmount)) +'!!'+FORMAT(ROUND(PRAmount))+'??';
                    Date:=CALCDATE('+1M', Date);
                     REPEAT
                       InterestRate:=Loansetup."Interest rate";
                       LPrincipal:=ROUND(LoanAmount/RepayPeriod,0.05,'>');
                     LInterest:=ROUND((InterestRate/12/100)*LoanAmount,0.05,'>');
                    TotalMRepay:=LPrincipal+LInterest;
                    PRAmount:=LoanAmount;
                      LoanAmount:=LoanAmount-LPrincipal;
                   RepayPeriod:= RepayPeriod-1;

                    text:=text+FORMAT(Date)+'!!'+FORMAT(ROUND( LPrincipal))+'!!'+FORMAT(ROUND( LInterest))+'!!'+FORMAT(ROUND(TotalMRepay))+'!!'+FORMAT(ROUND(LoanAmount)) +'!!'+FORMAT(ROUND(PRAmount))+'??';
                    Date:=CALCDATE('+1M', Date);
                    UNTIL RepayPeriod=0;
              END;





           END;
       }
    END;

    LOCAL PROCEDURE AutoPostLoanApplication@37(loanappcode@1000 : Integer) : Text;
    BEGIN
      OnlineLoanApplications.RESET;
      OnlineLoanApplications.SETRANGE("Loan App ID", loanappcode);
      IF OnlineLoanApplications.FIND('-') THEN BEGIN

        //Confirm Member Does not have existing loan with balance
        {{LoansRec.RESET;
        LoansRec.SETRANGE(LoansRec."BOSA No",OnlineLoanApplications."Member ID");
        LoansRec.SETRANGE(LoansRec."Loan Product Type",OnlineLoanApplications."Loan Type");
        IF LoansRec.FIND('-') THEN BEGIN
          REPEAT
            IF LoansRec."Loan Product Type"<>'GUR' THEN BEGIN
              LoansRec.CALCFIELDS(LoansRec."Outstanding Balance",LoansRec."Top Up Amount");
              IF (LoansRec."Outstanding Balance">1)THEN BEGIN
                EXIT('HASEXISTINGLOAN');
              END;
            END;
          UNTIL LoansRec.NEXT=0;
        END;}}

        //Start Posting Loan To loan Register
        LoansRec.RESET;
        LoansRec.INIT;
        LoansRec.Source := LoansRec.Source::BOSA;
        LoansRec."Mode of Disbursement" := LoansRec."Mode of Disbursement"::EFT;
        LoansRec.Discard := LoansRec.Discard::"1";
        LoansRec."BOSA No" := OnlineLoanApplications."Member ID";
        LoansRec."Client Code" := OnlineLoanApplications."Member ID";
        LoansRec."Requested Amount" := OnlineLoanApplications."Requested Amount";
        LoansRec.VALIDATE("Client Code");

        //Getting Loan Number
        SalesSetup.GET;
        SalesSetup.TESTFIELD(SalesSetup."BOSA Loans Nos");
        NoSeriesMgt.InitSeries(SalesSetup."BOSA Loans Nos",LoansRec."No. Series",0D,LoansRec."Loan  No.",LoansRec."No. Series");

        LoansRec."Application Date":=TODAY;
        LoansRec.Advice:=TRUE;
        LoansRec."Captured By":='PORTALADMIN';
        Users.RESET;
        Users.SETRANGE(Users."User Name",'PORTALADMIN');
        IF Users.FIND('-') THEN BEGIN
          LoansRec."Cashier Branch":=Users."Branch Code";
        END;

        LoansRec."Loan Product Type" := OnlineLoanApplications."Loan Type";
        LoansRec.VALIDATE("Loan  No.");
        LoansRec.VALIDATE("Loan Product Type");
        LoansRec.INSERT();

        AutoPostGuarantors(LoansRec."Loan  No.", OnlineLoanApplications."Loan App ID");
        OnlineLoanApplications."Loan Register Code" := LoansRec."Loan  No.";
        OnlineLoanApplications.Status := OnlineLoanApplications.Status::Processing;
        OnlineLoanApplications."Last Update Time" := CREATEDATETIME(TODAY,TIME);
        OnlineLoanApplications.MODIFY();
        EXIT('DONE');
      END;
      EXIT('NOTFOUND');
    END;

    LOCAL PROCEDURE AutoPostGuarantors@43(loannumber@1000 : Code[20];loannumberapp@1001 : Integer);
    VAR
      SelfGuaranteeAmount@1002 : Decimal;
    BEGIN
      OnlineLoanGuarantors.RESET;
      OnlineLoanGuarantors.SETRANGE("Loan Application ID", loannumberapp);
      OnlineLoanGuarantors.SETRANGE(Status, OnlineLoanGuarantors.Status::Approved);
      IF OnlineLoanGuarantors.FINDSET THEN BEGIN
        REPEAT
          objMember.GET(OnlineLoanGuarantors."Guarantor Id");

          LoanGuarantors.INIT;
          LoanGuarantors."Loan No" := loannumber;
          LoanGuarantors."Staff/Payroll No." := objMember."Personal No";
          LoanGuarantors."Member No" := objMember."No.";
          objMember.CALCFIELDS(TLoansGuaranteed,"Current Savings");
          LoanGuarantors."Shares *3":=(objMember."Current Savings");
          LoanGuarantors."TotalLoan Guaranteed":=objMember.TLoansGuaranteed;
          objMember.CALCFIELDS("Outstanding Balance","Current Shares","Total Committed Shares");
          LoanGuarantors.Name:=objMember.Name;
          LoanGuarantors."Staff/Payroll No." := objMember."Personal No";
          LoanGuarantors."Loan Balance":=objMember."Outstanding Balance";
          LoanGuarantors.Shares:=objMember."Current Shares"*1;
          LoanGuarantors."Amont Guaranteed" := OnlineLoanGuarantors."Guarantorship Amount";
          LoanGuarantors."TotalLoan Guaranteed":=objMember.TLoansGuaranteed;
          LoanGuarantors."Free Shares":=(objMember."Current Shares"*3)-objMember.TLoansGuaranteed;
          LoanGuarantors.UncommittedShares := objMember."Current Shares" - objMember."Total Committed Shares";
          LoanGuarantors.INSERT();
        UNTIL OnlineLoanGuarantors.NEXT = 0;
      END;
    END;

    PROCEDURE PollPendingEMAILS@39() MessageDetails : Text;
    VAR
      memberno@1000 : Text[30];
    BEGIN
          BEGIN


              SMSMessages.RESET;
              SMSMessages.SETRANGE(SMSMessages."Email Sent", SMSMessages."Email Sent"::No);
              SMSMessages.SETRANGE("Date Entered",TODAY);
              //SMSMessages.SetRange(SMSMessages."Entry No",898758);
              //SMSMessages.SetFilter(SMSMessages."Date Entered", '=%', TODAY);

              IF SMSMessages.FIND('-') THEN BEGIN

                  IF (SMSMessages."Telephone No" = '')
                    OR (SMSMessages."Telephone No" = '+')
                    OR (SMSMessages."SMS Message" = '')
                    THEN BEGIN

                      SMSMessages."Sent To Server" := SMSMessages."Email Sent"::Failed;

                      SMSMessages.MODIFY;

                  END ELSE BEGIN
                     memberno := FnGetMemberNo(SMSMessages."Telephone No");
                      Member.RESET;
                      Member.SETRANGE("No.", memberno);
                      IF Member.FIND('-') THEN BEGIN
                          MessageDetails := '';

                          MessageDetails += FORMAT(SMSMessages."Entry No") + '`' + Member."E-Mail" + '`' + FORMAT(SMSMessages."SMS Message") + '~';
                          SMSMessages."Email Sent" := SMSMessages."Email Sent"::Yes;
                          SMSMessages.MODIFY();
                          EXIT(MessageDetails);
                      END;
                      SMSMessages."Email Sent" := SMSMessages."Email Sent"::Failed;
                      //SMSMessages."Entry No." := 'FAILED';
                      SMSMessages.MODIFY;

                  END;
              END;
          END;
    END;

    PROCEDURE FnGetMemberNo@41(phoneNo@1000 : Code[100]) : Code[100];
    BEGIN
      BEGIN
          Member.RESET;
          Member.SETRANGE(Member."Mobile Phone No", phoneNo);
          IF Member.FIND('-') THEN
              EXIT(Member."No.");

          Member.RESET;
          Member.SETRANGE(Member."Phone No.", phoneNo);
          IF Member.FIND('-') THEN
              EXIT(Member."No.");

          Member.RESET;
          Member.SETRANGE(Member."Mobile Phone No", '0' + COPYSTR(phoneNo, 4, 15));
          IF Member.FIND('-') THEN
              EXIT(Member."No.");

          Member.RESET;
          Member.SETRANGE(Member."Phone No.", '0' + COPYSTR(phoneNo, 4, 15));
          IF Member.FIND('-') THEN
              EXIT(Member."No.");

          Member.RESET;
          Member.SETRANGE(Member."No.", phoneNo);
          IF Member.FIND('-') THEN
              EXIT(Member."No.");

          EXIT('');
      END;
    END;

    BEGIN
    END.
